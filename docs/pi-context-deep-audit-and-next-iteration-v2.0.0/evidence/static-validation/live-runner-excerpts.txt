   605	      margin: 0.02,
   606	    },
   607	    efficiency: {
   608	      tokenMedianRelativeDelta,
   609	      costPerSuccessRelativeDelta,
   610	      overflowRecovery: { B0: overflowB0, B1: overflowB1 },
   611	      overflowQuality,
   612	      realizedNetMedian,
   613	      budgetMismatchRate,
   614	    },
   615	    decision,
   616	    byFamily,
   617	    pairs: rows.map((row) => ({
   618	      id: row.id,
   619	      family: row.family,
   620	      seed: row.seed,
   621	      sameCut: row.sameCut,
   622	      expectedFirstKeptId: row.expectedFirstKeptId,
   623	      b0: slimArm(row.b0, [findSecret(row.id)]),
   624	      b1: slimArm(row.b1, [findSecret(row.id)]),
   625	    })),
   626	  };
   627	  const reportPath = join(outDir, "report.json");
   628	  writeFileSync(reportPath, `${JSON.stringify(report, null, 2)}\n`);
   629	  const digest = createHash("sha256").update(JSON.stringify(report)).digest("hex");
   630	  const gateDecision = {
   631	    gate: "w2-compactor",
   632	    decision,
   633	    hardGatePass,
   634	    publicationClaim,
   635	    livePiNative: true,
   636	    reasons: [
   637	      `profile=${profile} completed=${completed.length}/${expectedPairs} sameCut=${sameCut.length}`,
   638	      `B0 fromHook=false required; observed native=${String(b0Native)}`,
   639	      `B1 fromHook=true required; observed pcr=${String(b1FromHook)}`,
   640	      `maxTokens unmodified ${modelLimits.maxTokens}; keepRecentTokens=${LIVE_KEEP_RECENT_TOKENS} shared`,
   641	      `tokenMedianRelativeDelta=${tokenMedianRelativeDelta.toFixed(4)} realizedNetMedian=${realizedNetMedian}`,
   642	      sampleMeetsW2Gate ? "sample meets W2 100-pair floor" : "sample below W2 publication floor (100 pairs × 3 seeds)",
   643	    ],
   644	    reportHash: digest,
   645	    reportPath,
   646	  };
   647	  writeFileSync(join(outDir, "gate-decision.json"), `${JSON.stringify(gateDecision, null, 2)}\n`);
   648	  writeFileSync(
   649	    join(outDir, "run-manifest.json"),
   650	    `${JSON.stringify(
   651	      {
   652	        runId: report.runId,
   653	        profile,
   654	        generatedAt: report.generatedAt,
   655	        files: {
   656	          "report.json": digest,
   657	          "gate-decision.json": createHash("sha256").update(JSON.stringify(gateDecision)).digest("hex"),
   658	        },
   659	      },
   660	      null,
   661	      2,
   662	    )}\n`,
   663	  );
   664	  return { reportPath, decision, report };
   665	}
