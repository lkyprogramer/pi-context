     1  #!/usr/bin/env python3
     2  from __future__ import annotations
     3  import argparse, json
     4  from pathlib import Path
     5  root=Path(__file__).resolve().parents[1]
     6  index=json.loads((root/'tasks/TASK-INDEX.json').read_text())

     9      if state_path.exists(): return json.loads(state_path.read_text())
    10      return {t['id']:{'status':'pending','owner':None} for t in index}

    13  p=argparse.ArgumentParser(); sub=p.add_subparsers(dest='cmd',required=True)
    14  sub.add_parser('init'); sub.add_parser('next')
    15  c=sub.add_parser('claim'); c.add_argument('task'); c.add_argument('--owner',required=True)
    16  d=sub.add_parser('done'); d.add_argument('task'); d.add_argument('--owner',required=True)
    17  a=p.parse_args(); s=load()
    18  if a.cmd=='init': save(s); print(state_path)
    19  elif a.cmd=='next':
    20      rows=[t['id'] for t in index if ready(t,s)]; print('\n'.join(rows) if rows else 'none')
    21  elif a.cmd=='claim':
    22      t=next(x for x in index if x['id']==a.task)
    23      if not ready(t,s): raise SystemExit('not-ready')
    24      s[a.task]={'status':'claimed','owner':a.owner}; save(s); print(a.task)
    25  elif a.cmd=='done':
    26      if s[a.task].get('owner')!=a.owner: raise SystemExit('owner-mismatch')
    27      evidence=root/f'artifacts/task-evidence/{a.task}/evidence.json'
    28      if not evidence.exists(): raise SystemExit('missing-evidence')
    29      s[a.task]={'status':'done','owner':a.owner}; save(s); print(a.task)
