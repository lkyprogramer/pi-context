   571  export function evaluateW5Gate(input: {
   572    commit: string;
   573    headCommit: string;
   574    lanes: readonly W5LaneId[];
   575    familyRegressions?: readonly string[];
   576    bundles: Readonly<Record<string, unknown>>;
   577    signal?: AbortSignal;
   578  }): { decision: GateDecision["decision"]; publicationClaim: boolean; reasons: string[]; missing: string[] } {
   579    if (!input || typeof input !== "object") failMissing("input");
   580    if (input.signal !== undefined && !(input.signal instanceof AbortSignal)) failInput("signal");
   581    input.signal?.throwIfAborted();
   582    const required: W5LaneId[] = [
   583      "boundary",
   584      "natural-threshold",
   585      "overflow",
   586      "recursive",
   587      "fault-security",
   588      "performance-cache",
   589    ];
   590    const missing = required.filter((lane) => !input.lanes.includes(lane) || input.bundles[lane] == null);
   591    if (missing.length > 0) {
   592      return { decision: "stop", publicationClaim: false, reasons: ["missing-lane"], missing };
   593    }
   594    if (input.commit !== input.headCommit || !/^[a-f0-9]{40}$/u.test(input.commit)) {
   595      return { decision: "repeat-after-infrastructure-fix", publicationClaim: false, reasons: ["stale-commit"], missing: [] };
   596    }
   597    if ((input.familyRegressions ?? []).length > 0) {
   598      return { decision: "keep-pi-native", publicationClaim: false, reasons: ["family-regression"], missing: [] };
   599    }
   600    const engine = createGateEngine({
   601      workspaceId: "ws-report",
   602      git: { async status() { return { commit: input.commit, diffHash: EMPTY_DIFF, dirty: false }; } },
   603      files: { async mkdir() {}, async writeFile() {} },
   604    });
   605    const bundle = deriveW5RunBundle({
   606      commit: input.commit,
   607      bundles: input.bundles,
   608      familyRegressions: input.familyRegressions ?? [],
   609    });
   610    const decision = engine.evaluate(bundle);
   611    return {
   612      decision: decision.decision,
   613      publicationClaim: decision.decision === "adopt-pcr-compactor",
   614      reasons: [...decision.reasons],
   615      missing: [],
   616    };
   617  }
