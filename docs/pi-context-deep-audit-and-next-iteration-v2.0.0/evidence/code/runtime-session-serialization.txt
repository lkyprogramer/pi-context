   165    #validate(input: { operationId: string; cursor: RuntimeSessionScope; signal?: AbortSignal }): void {
   166      this.#assertOpen();
   167      validateOperation(input);
   168      const cursor = input.cursor;
   169      const mismatch =
   170        cursor.workspaceId !== this.scope.workspaceId ||
   171        cursor.sessionId !== this.scope.sessionId ||
   172        cursor.leafId !== this.scope.leafId ||
   173        cursor.lineageHash !== this.scope.lineageHash;
   174      if (mismatch) {
   175        throw new RuntimeSessionError("PCR_RUNTIME_SCOPE_MISMATCH", {
   176          expectedWorkspaceId: this.scope.workspaceId,
   177          expectedSessionId: this.scope.sessionId,
   178          expectedLeafId: this.scope.leafId,
   179          expectedLineageHash: this.scope.lineageHash,
   180        });
   181      }
   182    }

   184    async #exclusive<T>(work: () => Promise<T>): Promise<T> {
   185      let release!: () => void;
   186      const previous = this.#writeChain;
   187      this.#writeChain = new Promise<void>((resolve) => {
   188        release = resolve;
   189      });
   190      await previous;
   191      while (this.#reads > 0) await this.#readIdle;
   192      try {
   193        return await work();
   194      } finally {
   195        release();
   196      }
   197    }
   198  
   199    async #mutate<T>(
   200      input: { operationId: string; cursor: RuntimeSessionScope; signal?: AbortSignal },
   201      work: () => Promise<T>,
   202    ): Promise<T> {
   203      return this.#exclusive(async () => {
   204        this.#validate(input);
   205        return work();
   206      });
   207    }

   209    async #read<T>(
   210      input: { operationId: string; cursor: RuntimeSessionScope; signal?: AbortSignal },
   211      work: () => Promise<T>,
   212    ): Promise<T> {
   213      await this.#writeChain;
   214      this.#validate(input);
   215      if (this.#reads === 0) {
   216        this.#readIdle = new Promise<void>((resolve) => {
   217          this.#notifyReadsIdle = resolve;
   218        });
   219      }
   220      this.#reads += 1;
   221      try {
   222        return await work();
   223      } finally {
   224        this.#reads -= 1;
   225        if (this.#reads === 0) {
   226          this.#notifyReadsIdle?.();
   227          this.#notifyReadsIdle = undefined;
   228          this.#readIdle = Promise.resolve();
   229        }
   230      }
   231    }
