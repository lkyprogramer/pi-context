   366  async function runPair(item: W2Case, extensionPath: string): Promise<LivePairRow> {
   367    const root = mkdtempSync(join(tmpdir(), `pcr-w2-live-${item.id}-`));
   368    const cwd = join(root, "ws");
   369    mkdirSync(cwd, { recursive: true });
   370    const seedFile = join(root, "seed.jsonl");
   371    const frozen = writeW1ShapedSession({ sessionFile: seedFile, cwd, item });
   372    const b0File = join(root, "b0", "session.jsonl");
   373    const b1File = join(root, "b1", "session.jsonl");
   374    mkdirSync(dirname(b0File), { recursive: true });
   375    mkdirSync(dirname(b1File), { recursive: true });
   376    copyFileSync(seedFile, b0File);
   377    copyFileSync(seedFile, b1File);
   378    const agentB0 = join(root, "agent-b0");
   379    const agentB1 = join(root, "agent-b1");
   380    copyModelsUnmodified(agentB0);
   381    copyModelsUnmodified(agentB1);
   382    const [b0, b1] = await Promise.all([
   383      runArm({ arm: "B0", item, sessionFile: b0File, cwd, agentDir: agentB0, extensionPath }),
   384      runArm({ arm: "B1", item, sessionFile: b1File, cwd, agentDir: agentB1, extensionPath }),
   385    ]);
   386    rmSync(root, { recursive: true, force: true });
   387    return {
   388      id: item.id,
   389      family: item.family,
   390      seed: 0,
   391      sameCut: Boolean(b0.firstKeptEntryId && b0.firstKeptEntryId === b1.firstKeptEntryId),
   392      expectedFirstKeptId: frozen.expectedFirstKeptId,
   393      b0,
   394      b1,
   395    };
   396  }

   398  function redact(text: string, secrets: string[]): string {
   399    let out = text.slice(0, 400);
   400    for (const secret of secrets) out = out.replaceAll(secret, "[redacted]");
   401    return out;
