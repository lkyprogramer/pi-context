```ts
import { expect, test } from "vitest";
 import { RequestWitnessTracker } from "../../src/projection/witness.js";
 test("another session cannot confirm an original", () => {
   const id = { workspaceId:"w",sessionId:"s",provider:"p",model:"m",
     configHash:"c",compactionBoundary:null,epoch:1 };
   const tracker = new RequestWitnessTracker();
   const requestId = tracker.prepare(id, new Map([["r:0","hash"]]), "view");
   tracker.markSent(requestId, id, ["r:0"]);
   expect(tracker.accept(requestId,{...id,sessionId:"other"},"stop","resp-1")).toBe(false);
   expect(tracker.has(id,"r:0","hash")).toBe(false);
   expect(tracker.accept(requestId,id,"stop","resp-1")).toBe(true);
   expect(tracker.has(id,"r:0","hash")).toBe(true);
 });
```

## 最小实施顺序

1. 实现有界内存Tracker，只记录本次原文fieldHashes和身份，不保存文本副本；reset不回退epoch，作用域变更统一+1。
2. context输出后prepare原文仍可见的字段。before_provider_request对已支持OpenAI tool message仅只读校验callID与最终content，成功markSent；未知payload方法标unavailable且不新增原文暴露。
3. 只在对应assistant成功stop/toolUse且响应可识别时accept。error/aborted/length/缺pending/user/tool消息均不确认；多次HTTP回调不凭相同输入Hash合并真正请求。
4. profile切换调用configHashOf并fence；provider+model完整身份；成功session_compact不管willRetry均清plan/pending，session_tree/reload同理。正常追加不抹去已确认的祖先。
5. 先render新candidate计算actual applied delta，再决定发布、recordFold。applied0不能记fold或saved；记录本地applied与sent/confirmed为不同状态。对renderHash失配不应用对应字段。
6. 受控Provider分三轮：第一轮原文、第二轮无成功响应不折叠、成功后移出保护带且压力达标第三轮实际变短；restart后完整发送一次再允许折叠。

## 负例与边界

- P02/P03不再用全日志成功assistant替原文确认。
- 模型同id但不同provider、配置变更、overflow compact正确fence。
- 未知Provider仍有observe/history，不能因无法wire验证而破坏请求。
- 不把“发送成功”写成“理解无损”。
